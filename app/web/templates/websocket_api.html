How to Connect from JavaScript (Web / Node):
const socket = new WebSocket("ws://localhost:8000/ws");

socket.onopen = () => {
  // Query any GET API:
  socket.send(JSON.stringify({
    action: "get_files",
    params: { limit: 50, query: "2609" }
  }));
};

socket.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  console.log("Live API Data:", msg);
};
How to Connect from Python (asyncio + websockets):
import asyncio
import json
import websockets

async def listen():
    async with websockets.connect("ws://localhost:8000/ws") as ws:
        # Request Duplicate Groups
        await ws.send(json.dumps({"action": "get_duplicates"}))
        
        while True:
            message = await ws.recv()
            data = json.loads(message)
            print("Received Live Data:", data.get("type"))

asyncio.run(listen())
